function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var action = e.parameter.action; // Handler Khusus Data Monitoring Real-Time untuk Superior & Admin if (action === "getMonitoring") { var sheetAbsensi = ss.getSheetByName("Sheet1") || ss.getSheets()[0]; var dataAbsensi = sheetAbsensi.getDataRange().getValues(); var sheetKaryawan = ss.getSheetByName("Data_Karyawan"); var dataKaryawan = sheetKaryawan ? sheetKaryawan.getDataRange().getValues() : []; var zonaWaktu = "Asia/Jakarta"; var hariIniStr = Utilities.formatDate(new Date(), zonaWaktu, "yyyy-MM-dd"); var listAbsenHariIni = []; for (var i = 1; i < dataAbsensi.length; i++) { var row = dataAbsensi[i]; var timestampRaw = row[0]; var tglStr = ""; if (timestampRaw instanceof Date) { tglStr = Utilities.formatDate(timestampRaw, zonaWaktu, "yyyy-MM-dd"); } else if (typeof timestampRaw === "string") { tglStr = timestampRaw.split(" ")[0]; } if (tglStr === hariIniStr) { listAbsenHariIni.push({ timestamp: row[0], nik: String(row[1]), nama: row[2], tipe: row[3], shift: row[4], toko: row[5], foto: row[6], gps: row[7], keterangan: row[8] }); } } var masterKaryawan = []; for (var j = 1; j < dataKaryawan.length; j++) { var n = String(dataKaryawan[j][0]).trim(); var nm = String(dataKaryawan[j][1]).trim(); if (n && nm) { masterKaryawan.push({ nik: n, nama: nm }); } } return ContentService.createTextOutput(JSON.stringify({ status: "success", tanggal: hariIniStr, absenToday: listAbsenHariIni, totalKaryawan: masterKaryawan })).setMimeType(ContentService.MimeType.JSON); } // Default Load Data Toko & Master Karyawan var sheetToko = ss.getSheetByName("Daftar_Toko"); var daftarToko = []; if (sheetToko) { var dataToko = sheetToko.getDataRange().getValues(); for (var k = 1; k < dataToko.length; k++) { var kodeToko = String(dataToko[k][0]).trim(); var lat = parseFloat(dataToko[k][2]); var lng = parseFloat(dataToko[k][3]); if (kodeToko && !isNaN(lat) && !isNaN(lng)) { daftarToko.push({ kode: kodeToko, latitude: lat, longitude: lng }); } } } var sheetKaryawanInit = ss.getSheetByName("Data_Karyawan"); var daftarKaryawan = {}; if (sheetKaryawanInit) { var dK = sheetKaryawanInit.getDataRange().getValues(); for (var m = 1; m < dK.length; m++) { var nikStr = String(dK[m][0]).trim(); var namaStr = String(dK[m][1]).trim(); if (nikStr && namaStr) { daftarKaryawan[nikStr] = namaStr; } } } return ContentService.createTextOutput(JSON.stringify({ status: "success", toko: daftarToko, karyawan: daftarKaryawan })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
